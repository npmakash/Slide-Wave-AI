// =====================================================
// SETTINGS
// =====================================================

const SHEET_NAME = "Reasoning_Qs";

const PRESENTATION_ID = "1uvut1VsI22SSQNoV0BWkFVbRG8HwW25XhHG4Pc6PGqQ";

// Template slide:
// First slide = 0
// Second slide = 1
// Third slide = 2
const TEMPLATE_SLIDE_INDEX = 0;

// Questions per slide
const QUESTIONS_PER_SLIDE = 12;


// =====================================================
// MAIN FUNCTION
// RUN ONLY THIS FUNCTION
// =====================================================

function generateQuestionSlides() {

  Logger.log("STARTING...");


  // ---------------------------------------------------
  // 1. GET GOOGLE SHEET
  // ---------------------------------------------------

  const spreadsheet =
    SpreadsheetApp.getActiveSpreadsheet();

  const sheet =
    spreadsheet.getSheetByName(SHEET_NAME);


  if (!sheet) {

    throw new Error(
      "Sheet not found: " + SHEET_NAME
    );

  }


  // ---------------------------------------------------
  // 2. READ SHEET
  // ---------------------------------------------------

  const data =
    sheet.getDataRange().getValues();


  if (data.length < 2) {

    throw new Error(
      "No question data found in the sheet."
    );

  }


  // ---------------------------------------------------
  // 3. READ HEADERS
  // ---------------------------------------------------

  const headers =
    data[0].map(function(header) {

      return String(header)
        .trim()
        .toLowerCase();

    });


  Logger.log(
    "Headers: " + headers.join(", ")
  );


  // ---------------------------------------------------
  // 4. FIND COLUMNS
  // ---------------------------------------------------

  const numberColumn =
    headers.indexOf("number");

  const questionColumn =
    headers.indexOf("question");

  const optionAColumn =
    headers.indexOf("optiona");

  const optionBColumn =
    headers.indexOf("optionb");

  const optionCColumn =
    headers.indexOf("optionc");

  const optionDColumn =
    headers.indexOf("optiond");

  const answerColumn =
    headers.indexOf("answer");

  const explanationColumn =
    headers.indexOf("explanation");


  // ---------------------------------------------------
  // 5. CHECK REQUIRED COLUMNS
  // ---------------------------------------------------

  if (
    numberColumn === -1 ||
    questionColumn === -1 ||
    optionAColumn === -1 ||
    optionBColumn === -1 ||
    optionCColumn === -1 ||
    optionDColumn === -1
  ) {

    throw new Error(
      "Required columns not found.\n\n" +
      "Your first row must contain:\n\n" +
      "number\n" +
      "question\n" +
      "optionA\n" +
      "optionB\n" +
      "optionC\n" +
      "optionD\n\n" +
      "Optional:\n" +
      "answer\n" +
      "explanation"
    );

  }


  // ---------------------------------------------------
  // 6. CREATE QUESTIONS ARRAY
  // ---------------------------------------------------

  const questions = [];


  for (
    let row = 1;
    row < data.length;
    row++
  ) {

    const currentRow =
      data[row];


    // Ignore completely empty rows
    let isEmpty = true;


    for (
      let c = 0;
      c < currentRow.length;
      c++
    ) {

      if (
        currentRow[c] !== "" &&
        currentRow[c] !== null
      ) {

        isEmpty = false;
        break;

      }

    }


    if (isEmpty) {
      continue;
    }


    // Create question object

    const question = {

      number:
        currentRow[numberColumn],

      question:
        currentRow[questionColumn],

      optionA:
        currentRow[optionAColumn],

      optionB:
        currentRow[optionBColumn],

      optionC:
        currentRow[optionCColumn],

      optionD:
        currentRow[optionDColumn],

      answer:
        answerColumn !== -1
          ? currentRow[answerColumn]
          : "",

      explanation:
        explanationColumn !== -1
          ? currentRow[explanationColumn]
          : ""

    };


    questions.push(question);

  }


  // ---------------------------------------------------
  // 7. CHECK QUESTIONS
  // ---------------------------------------------------

  if (questions.length === 0) {

    throw new Error(
      "No questions found."
    );

  }


  Logger.log(
    "Total questions: " +
    questions.length
  );


  // ---------------------------------------------------
  // 8. OPEN GOOGLE SLIDES
  // ---------------------------------------------------

  const presentation =
    SlidesApp.openById(
      PRESENTATION_ID
    );


  // ---------------------------------------------------
  // 9. GET TEMPLATE SLIDE
  // ---------------------------------------------------

  let slides =
    presentation.getSlides();


  if (
    TEMPLATE_SLIDE_INDEX < 0 ||
    TEMPLATE_SLIDE_INDEX >= slides.length
  ) {

    throw new Error(
      "Template slide index is invalid."
    );

  }


  const template =
    slides[TEMPLATE_SLIDE_INDEX];


  Logger.log(
    "Template slide found."
  );


  // ---------------------------------------------------
  // 10. DELETE OLD GENERATED SLIDES
  //
  // Keep ONLY the template slide.
  // ---------------------------------------------------

  slides =
    presentation.getSlides();


  for (
    let i = slides.length - 1;
    i >= 0;
    i--
  ) {

    if (
      i !== TEMPLATE_SLIDE_INDEX
    ) {

      slides[i].remove();

    }

  }


  Logger.log(
    "Old slides removed."
  );


  // ---------------------------------------------------
  // 11. CALCULATE NUMBER OF SLIDES
  // ---------------------------------------------------

  const totalPages =
    Math.ceil(
      questions.length /
      QUESTIONS_PER_SLIDE
    );


  Logger.log(
    "Pages required: " +
    totalPages
  );


  // ---------------------------------------------------
  // 12. CREATE SLIDES IN CORRECT ORDER
  //
  // Template = slide 1
  //
  // Page 1 = slide 2
  // Page 2 = slide 3
  // Page 3 = slide 4
  // etc.
  // ---------------------------------------------------

  for (
    let page = 0;
    page < totalPages;
    page++
  ) {


    const startIndex =
      page * QUESTIONS_PER_SLIDE;


    const endIndex =
      Math.min(
        startIndex + QUESTIONS_PER_SLIDE,
        questions.length
      );


    const pageQuestions =
      questions.slice(
        startIndex,
        endIndex
      );


    Logger.log(
      "Creating slide " +
      (page + 1) +
      " with questions " +
      (startIndex + 1) +
      " to " +
      endIndex
    );


    // -------------------------------------------------
    // CREATE COPY DIRECTLY AT CORRECT POSITION
    //
    // index 0 = template
    // index 1 = questions 1-12
    // index 2 = questions 13-24
    // etc.
    // -------------------------------------------------

    const newSlide =
      presentation.insertSlide(
        page + 1,
        template
      );


    // -------------------------------------------------
    // FILL THIS SLIDE
    // -------------------------------------------------

    fillQuestionSlide(
      newSlide,
      pageQuestions
    );

  }


  // ---------------------------------------------------
  // 13. SAVE
  // ---------------------------------------------------

  presentation.saveAndClose();


  Logger.log(
    "===================================="
  );

  Logger.log(
    "SUCCESS"
  );

  Logger.log(
    "Questions: " +
    questions.length
  );

  Logger.log(
    "Generated slides: " +
    totalPages
  );

  Logger.log(
    "===================================="
  );

}


// =====================================================
// FILL ONE SLIDE
// =====================================================

function fillQuestionSlide(
  slide,
  questions
) {


  // There are always 12 positions
  // on the template.

  for (
    let position = 1;
    position <= QUESTIONS_PER_SLIDE;
    position++
  ) {


    const question =
      questions[position - 1];


    // -------------------------------------------------
    // QUESTION EXISTS
    // -------------------------------------------------

    if (question) {


      replacePlaceholder(
        slide,
        "{{number_" + position + "}}",
        question.number
      );


      replacePlaceholder(
        slide,
        "{{question_" + position + "}}",
        question.question
      );


      replacePlaceholder(
        slide,
        "{{optionA_" + position + "}}",
        question.optionA
      );


      replacePlaceholder(
        slide,
        "{{optionB_" + position + "}}",
        question.optionB
      );


      replacePlaceholder(
        slide,
        "{{optionC_" + position + "}}",
        question.optionC
      );


      replacePlaceholder(
        slide,
        "{{optionD_" + position + "}}",
        question.optionD
      );


      replacePlaceholder(
        slide,
        "{{answer_" + position + "}}",
        question.answer
      );


      replacePlaceholder(
        slide,
        "{{explanation_" + position + "}}",
        question.explanation
      );


    }

    // -------------------------------------------------
    // NO QUESTION
    //
    // Important for last slide.
    // -------------------------------------------------

    else {


      replacePlaceholder(
        slide,
        "{{number_" + position + "}}",
        ""
      );


      replacePlaceholder(
        slide,
        "{{question_" + position + "}}",
        ""
      );


      replacePlaceholder(
        slide,
        "{{optionA_" + position + "}}",
        ""
      );


      replacePlaceholder(
        slide,
        "{{optionB_" + position + "}}",
        ""
      );


      replacePlaceholder(
        slide,
        "{{optionC_" + position + "}}",
        ""
      );


      replacePlaceholder(
        slide,
        "{{optionD_" + position + "}}",
        ""
      );


      replacePlaceholder(
        slide,
        "{{answer_" + position + "}}",
        ""
      );


      replacePlaceholder(
        slide,
        "{{explanation_" + position + "}}",
        ""
      );

    }

  }

}


// =====================================================
// REPLACE PLACEHOLDER
// =====================================================

function replacePlaceholder(
  slide,
  placeholder,
  value
) {


  if (
    value === null ||
    value === undefined
  ) {

    value = "";

  }


  slide.replaceAllText(
    placeholder,
    String(value)
  );

}


function removeUnusedQuestionSlots(slide, usedQuestions) {

  for (
    let position = usedQuestions + 1;
    position <= QUESTIONS_PER_SLIDE;
    position++
  ) {

    const placeholders = [
      "{{number_" + position + "}}",
      "{{question_" + position + "}}",
      "{{optionA_" + position + "}}",
      "{{optionB_" + position + "}}",
      "{{optionC_" + position + "}}",
      "{{optionD_" + position + "}}",
      "{{answer_" + position + "}}",
      "{{explanation_" + position + "}}"
    ];

    const elements = slide.getPageElements();

    elements.forEach(function(element) {

      if (element.getPageElementType() !== SlidesApp.PageElementType.SHAPE) {
        return;
      }

      const shape = element.asShape();
      const text = shape.getText().asString();

      for (let i = 0; i < placeholders.length; i++) {

        if (text.indexOf(placeholders[i]) !== -1) {
          shape.remove();
          break;
        }

      }

    });

  }

}
